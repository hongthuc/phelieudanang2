<script>
  // 1. Tự động trượt 2 Banner kích thước bằng nhau
  let currentSlide = 0;
  const slides = document.querySelectorAll('.banner-slide');
  const dots = document.querySelectorAll('.banner-dot');
  let slideInterval = null;

  function showSlide(index) {
    if (!slides.length) return;
    currentSlide = (index + slides.length) % slides.length;
    slides.forEach((slide, i) => {
      if (i === currentSlide) {
        slide.classList.remove('hidden');
        slide.classList.add('block');
      } else {
        slide.classList.remove('block');
        slide.classList.add('hidden');
      }
    });
    dots.forEach((dot, i) => {
      if (i === currentSlide) {
        dot.className = 'banner-dot w-6 h-2 rounded-full bg-brand-accentLight transition-all';
      } else {
        dot.className = 'banner-dot w-2.5 h-2 rounded-full bg-white/60 transition-all';
      }
    });
  }

  function changeSlide(step) {
    showSlide(currentSlide + step);
    resetSlideTimer();
  }

  function goToSlide(index) {
    showSlide(index);
    resetSlideTimer();
  }

  function resetSlideTimer() {
    if (slideInterval) clearInterval(slideInterval);
    slideInterval = setInterval(() => {
      showSlide(currentSlide + 1);
    }, 5000);
  }

  // 2. Menu trên điện thoại
  const mobileMenuBtn = document.getElementById('mobileMenuBtn');
  const mobileMenuDrawer = document.getElementById('mobileMenuDrawer');
  if (mobileMenuBtn && mobileMenuDrawer) {
    mobileMenuBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      mobileMenuDrawer.classList.toggle('hidden');
    });
    document.querySelectorAll('.mobile-nav-item').forEach(link => {
      link.addEventListener('click', () => {
        mobileMenuDrawer.classList.add('hidden');
      });
    });
  }

  // 3. Xử lý gửi Form Yêu cầu và Báo giá trên trang chủ (không dùng alert)
  function handleQuoteSubmit(event) {
    event.preventDefault();
    const name = document.getElementById('qName').value.trim();
    const phone = document.getElementById('qPhone').value.trim();
    const category = document.getElementById('qCategory').value;
    const address = document.getElementById('qAddress').value.trim();
    const box = document.getElementById('quoteSuccessBox');

    if (box) {
      box.classList.remove('hidden');
      box.innerHTML = 'Đã ghi nhận yêu cầu báo giá của <strong>' + name + '</strong> (' + phone + ') cho danh mục <strong>' + category + '</strong>' + (address ? ' tại ' + address : '') + '. Đội ngũ <strong>phelieudanang.com (0388.670.036)</strong> sẽ liên hệ lại ngay!';
    }
    document.getElementById('quoteRequestForm').reset();
  }

  window.onload = function () {
    showSlide(0);
    resetSlideTimer();
  };
